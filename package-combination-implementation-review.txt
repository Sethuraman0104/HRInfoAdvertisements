  135:                     await _context.AdvertisementEnquiries.CountAsync(x =>
  136:                         x.AdvertisementID == advertisement.AdvertisementID &&
  137:                         x.CreatedDate >= monthStart &&
  138:                         x.CreatedDate < nextMonthStart);
  139: 
  140:                 if (advertisementCount >=
  141:                     ownerPackage.MonthlyAdvertisementEnquiryLimit)
  142:                 {
  143:                     rejectionType = "ADVERTISEMENT_MONTHLY_LIMIT";
  144:                     rejectionCount = advertisementCount;
  145:                     rejectionLimit =
  146:                         ownerPackage.MonthlyAdvertisementEnquiryLimit;
  147:                     rejectionPackageCode = ownerPackage.PackageCode;
  148:                 }
  149:             }
  150: 
  151:             if (rejectionType == null)
  152:             {
  153:                 var enquiry = new AdvertisementEnquiry
  154:                 {
  155:                     AdvertisementID = advertisement.AdvertisementID,
  156:                     RecipientUserID = advertisement.UserID,
  157:                     SenderUserID = senderUserId,
  158:                     Subject = request.Subject.Trim(),
  159:                     Message = request.Message.Trim(),
  160:                     ContactMobile =
  161:                         string.IsNullOrWhiteSpace(request.ContactMobile)
  162:                             ? sender.MobileNo
  163:                             : request.ContactMobile.Trim(),
  164:                     ContactEmail =
  165:                         string.IsNullOrWhiteSpace(request.ContactEmail)
  166:                             ? sender.Email
  167:                             : request.ContactEmail.Trim(),
  168:                     Status = "OPEN",
  169:                     CreatedDate = DateTime.UtcNow
  170:                 };
  171:

  184:                             enquiry.AdvertisementEnquiryID,

  186:                         SenderUserID = senderUserId,
  187:                         RecipientUserID = advertisement.UserID,
  188:                         Subject = enquiry.Subject,
  189:                         CreatedDate = enquiry.CreatedDate
  190:                     }),
  191:                     CreatedDate = DateTime.UtcNow
  192:                 });
  193: 
  194:                 await _context.SaveChangesAsync();
  195:                 await transaction.CommitAsync();
  196: 
  197:                 transactionCompleted = true;
  198: createdEnquiryId = enquiry.AdvertisementEnquiryID;
  199: 
  200:                 return await BuildResponseAsync(
  201:                     enquiry.AdvertisementEnquiryID)
  202:                     ?? throw new InvalidOperationException(
  203:                         "Unable to retrieve the created enquiry.");
  204:             }
  205: 
  206:             // Release the limit locks before persisting the rejection audit.
  207:             await transaction.RollbackAsync();
  208:             transactionCompleted = true;
  209:         }
  210:         catch
  211:         {
  212:             if (!transactionCompleted)
  213:     {
  214:         try
  215:         {
  216:             await transaction.RollbackAsync();
  217:         }
  218:         catch
  219:         {
  220:             // Preserve the original exception.
  221:         }
  222:     }
  223: 
  224:             throw;
  225:         }
  226:     }
  227: 
  228:     if (createdEnquiryId.HasValue)
  229: {
  230:     return await BuildResponseAsync(createdEnquiryId.Value)
  231:         ?? throw new InvalidOperationException(
  232:             "The enquiry was created, but its response could not be retrieved.");
  233: }
  234: 
  235:     // Keep rejected-limit audit records outside the rolled-back transaction.
  236:     await WriteEnquiryLimitAuditAsync(
  237:         senderUserId,

  246:     {

  248:             $"Your monthly enquiry limit of {rejectionLimit} has been reached.");

  349:     public async Task<AdvertisementEnquiryMessageResponse?> ReplyAsync(
  350:     long userId,
  351:     long enquiryId,
  352:     ReplyAdvertisementEnquiryRequest request)
  353: {
  354:     if (string.IsNullOrWhiteSpace(request.Message))
  355:         throw new ArgumentException("Message is required.");
  356: 
  357:     var enquiry = await _context.AdvertisementEnquiries
  358:         .FirstOrDefaultAsync(x =>
  359:             x.AdvertisementEnquiryID == enquiryId &&
  360:             (x.SenderUserID == userId ||
  361:              x.RecipientUserID == userId));
  362: 
  363:     if (enquiry == null)
  364:         return null;
  365: 
  366:     if (string.Equals(
  367:         enquiry.Status,
  368:         "CLOSED",
  369:         StringComparison.OrdinalIgnoreCase))
  370:     {
  371:         throw new InvalidOperationException(
  372:             "This enquiry is closed and cannot receive new replies.");
  373:     }
  374: 
  375:     var now = DateTime.UtcNow;
  376:     var monthStart = new DateTime(
  377:         now.Year, now.Month, 1, 0, 0, 0, DateTimeKind.Utc);
  378:     var nextMonthStart = monthStart.AddMonths(1);
  379:     var monthKey = monthStart.ToString("yyyyMM");
  380: 
  381:     var replyLock =
  382:         $"HRInfoAdvertisements:ReplyLimit:User:{userId}:{monthKey}";
  383: 
  384:     string? packageCode = null;
  385:     var currentCount = 0;
  386:     var configuredLimit = 0;
  387:     var limitReached = false;
  388:     AdvertisementEnquiryMessage? savedMessage = null;
  389: 
  390:     await using (var transaction =
  391:         await _context.Database.BeginTransactionAsync())
  392:     {
  393:         var transactionCompleted = false;
  394: 
  395:         try
  396:         {
  397:             await AcquireEnquiryLimitLockAsync(replyLock);
  398:             await PackageEntitlementActivation.ActivateDueAsync(_context, userId, now);
  399: 
  400:             var package = await ResolveEnquiryPackageAsync(userId, now);
  401:             packageCode = package.PackageCode;
  402:             configuredLimit = package.MonthlyReplyLimit;
  403: 
  404:             currentCount = await _context.AdvertisementEnquiryMessages
  405:                 .CountAsync(x =>
  406:                     x.SenderUserID == userId &&
  407:                     x.CreatedDate >= monthStart &&
  408:                     x.CreatedDate < nextMonthStart);
  409: 
  410:             if (currentCount >= configuredLimit)
  411:             {
  412:                 limitReached = true;
  413: 
  414:                 await transaction.RollbackAsync();
  415:                 transactionCompleted = true;
  416:             }
  417:             else
  418:             {
  419:                 var message = new AdvertisementEnquiryMessage
  420:                 {
  421:                     AdvertisementEnquiryID =
  422:                         enquiry.AdvertisementEnquiryID,
  423:                     SenderUserID = userId,
  424:                     Message = request.Message.Trim(),
  425:                     CreatedDate = now,
  426:                     IsRead = false
  427:                 };
  428: 
  429:                 _context.AdvertisementEnquiryMessages.Add(message);
  430: 
  431:                 enquiry.LastRepliedDate = now;
  432:                 enquiry.Status = "OPEN";
  433: 
  434:                 await _context.SaveChangesAsync();
  435:                 await transaction.CommitAsync();
  436: 
  437:                 transactionCompleted = true;
  438:                 savedMessage = message;
  439:             }
  440:         }
  441:         catch
  442:         {
  443:             if (!transactionCompleted)
  444:             {
  445:                 try
  446:                 {
  447:                     await transaction.RollbackAsync();
  448:                 }
  449:                 catch
  450:                 {
  451:                     // Preserve the original exception.
  452:                 }
  453:             }
  454: 
  455:             throw;
  456:         }
  457:     }
  458: 
  459:     if (limitReached)
  460:     {
  461:         // Persist the audit record after releasing the limit lock.
  462:         _context.AuditLogs.Add(new AuditLog
  463:         {
  464:             UserID = userId,
  465:             Action = "REPLY_LIMIT_REJECTED",
  466:             EntityName = "AdvertisementEnquiry",
  467:             EntityID = enquiryId.ToString(),
  468:             NewValues = JsonSerializer.Serialize(new
  469:             {
  470:                 UserID = userId,
  471:                 AdvertisementEnquiryID = enquiryId,
  472:                 LimitType = "USER_MONTHLY_REPLY_LIMIT",
  473:                 CurrentCount = currentCount,
  474:                 ConfiguredLimit = configuredLimit,
  475:                 PackageCode = packageCode,
  476:                 MonthStartUtc = monthStart
  477:             }),
  478:             CreatedDate = DateTime.UtcNow
  479:         });
  480: 
  481:         await _context.SaveChangesAsync();
  482: 
  483:         throw new InvalidOperationException(
  484:             $"Your monthly reply limit of {configuredLimit} has been reached.");
  485:     }
  486: 
  487:     var sender = await _context.Users
  488:         .AsNoTracking()
  489:         .FirstOrDefaultAsync(x => x.UserID == userId);
  490: 
  491:     return new AdvertisementEnquiryMessageResponse
  492:     {
  493:         AdvertisementEnquiryMessageID =
  494:             savedMessage!.AdvertisementEnquiryMessageID,
  495:         AdvertisementEnquiryID =
  496:             savedMessage.AdvertisementEnquiryID,
  497:         SenderUserID = savedMessage.SenderUserID,
  498:         SenderName = GetUserDisplayName(sender),
  499:         Message = savedMessage.Message,
  500:         CreatedDate = savedMessage.CreatedDate,
  501:         IsRead = savedMessage.IsRead,
  502:         ReadDate = savedMessage.ReadDate

  622: private async Task AcquireEnquiryLimitLockAsync(
  623:     string resource)
  624: {
  625:     var transaction = _context.Database.CurrentTransaction
  626:         ?? throw new InvalidOperationException(
  627:             "An active database transaction is required for enquiry limit checks.");
  628: 
  629:     await using var command =
  630:         _context.Database.GetDbConnection().CreateCommand();
  631: 
  632:     command.Transaction = transaction.GetDbTransaction();
  633:     command.CommandType = CommandType.Text;
  634:     command.CommandText = """
  635:         DECLARE @result int;
  636: 
  637:         EXEC @result = sys.sp_getapplock
  638:             @Resource = @resource,
  639:             @LockMode = 'Exclusive',
  640:             @LockOwner = 'Transaction',
  641:             @LockTimeout = 10000,
  642:             @DbPrincipal = 'public';
  643: 
  644:         SELECT @result;
  645:         """;
  646: 
  647:     var parameter = command.CreateParameter();
  648:     parameter.ParameterName = "@resource";
  649:     parameter.DbType = DbType.String;
  650:     parameter.Size = 255;
  651:     parameter.Value = resource;
  652:     command.Parameters.Add(parameter);
  653: 
  654:     var result = await command.ExecuteScalarAsync();
  655:     var lockResult = Convert.ToInt32(result);
  656: 
  657:     // SQL Server returns a negative value if the lock
  658:     // cannot be acquired. Never continue without the lock.
  659:     if (lockResult < 0)
  660:     {
  661:         throw new InvalidOperationException(
  662:             "The enquiry limit could not be checked because of concurrent activity. Please try again.");
  663:     }
  664: }
