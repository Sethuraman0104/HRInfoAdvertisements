using HRInfoAdvertisements.Domain.Billing;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata.Builders;

namespace HRInfoAdvertisements.Infrastructure.Data.Configurations;

public class UserPackageEntitlementConfiguration
    : IEntityTypeConfiguration<UserPackageEntitlement>
{
    public void Configure(EntityTypeBuilder<UserPackageEntitlement> builder)
    {
        builder.ToTable(
            "UserPackageEntitlements",
            table =>
            {
                table.HasCheckConstraint(
                    "CK_UserPackageEntitlements_DateRange",
                    "[EndDate] IS NULL OR [EndDate] > [StartDate]");

                table.HasCheckConstraint(
                    "CK_UserPackageEntitlements_Status",
                    "[Status] IN ('PENDING', 'ACTIVE', 'CANCELLED', 'EXPIRED')");

                table.HasCheckConstraint(
                    "CK_UserPackageEntitlements_NonNegativeLimits",
                    "[MonthlyAdvertisementPostLimit] >= 0 AND " +
                    "[MonthlySenderEnquiryLimit] >= 0 AND " +
                    "[MonthlyAdvertisementEnquiryLimit] >= 0 AND " +
                    "[MonthlyReplyLimit] >= 0");
            });

        builder.HasKey(x => x.UserPackageEntitlementID);

        builder.Property(x => x.MonthlyAdvertisementPostLimit)
            .HasDefaultValue(0)
            .IsRequired();

        builder.Property(x => x.MonthlySenderEnquiryLimit)
            .HasDefaultValue(0)
            .IsRequired();

        builder.Property(x => x.MonthlyAdvertisementEnquiryLimit)
            .HasDefaultValue(0)
            .IsRequired();

        builder.Property(x => x.MonthlyReplyLimit)
            .HasDefaultValue(0)
            .IsRequired();

        builder.Property(x => x.Status)
            .HasMaxLength(20)
            .IsUnicode(false)
            .IsRequired();

        builder.HasIndex(x => new
        {
            x.UserID,
            x.Status,
            x.StartDate,
            x.EndDate
        });

        // Only one entitlement marked ACTIVE per user.
        builder.HasIndex(x => x.UserID)
            .IsUnique()
            .HasFilter("[Status] = 'ACTIVE'");

        builder.HasOne(x => x.User)
            .WithMany()
            .HasForeignKey(x => x.UserID)
            .OnDelete(DeleteBehavior.Restrict);

        builder.HasOne(x => x.EnquiryPackage)
            .WithMany(x => x.Entitlements)
            .HasForeignKey(x => x.EnquiryPackageID)
            .OnDelete(DeleteBehavior.Restrict);
    }
}

    public async Task<MyPackageBalanceResponse> GetMyPackageBalanceAsync(
        long userId,
        CancellationToken cancellationToken = default)
    {
        if (userId <= 0)
        {
            throw new ArgumentOutOfRangeException(nameof(userId));
        }

        var now = DateTime.UtcNow;
        var periodStart = new DateTime(
            now.Year, now.Month, 1, 0, 0, 0, DateTimeKind.Utc);
        var periodEnd = periodStart.AddMonths(1);

        var response = new MyPackageBalanceResponse
        {
            UsagePeriodStartUtc = periodStart,
            UsagePeriodEndUtc = periodEnd
        };

        var paidEntitlement = await _context.UserPackageEntitlements
            .AsNoTracking()
            .Where(x =>
                x.UserID == userId &&
                x.Status == UserPackageEntitlementStatuses.Active &&
                x.StartDate <= now &&
                (x.EndDate == null || x.EndDate > now) &&
                x.EnquiryPackage.PackageCode != "FREE" &&
                x.Payments.Any(p =>
                    p.Status == PaymentTransactionStatuses.Confirmed &&
                    p.ConfirmedDate != null &&
                    p.RefundedDate == null))
            .OrderByDescending(x => x.StartDate)
            .Select(x => new
            {
                x.EnquiryPackage.PackageCode,
                x.EnquiryPackage.Name,
                x.Status,
                x.StartDate,
                x.EndDate,
                x.MonthlyAdvertisementPostLimit,
                x.MonthlySenderEnquiryLimit,
                x.MonthlyAdvertisementEnquiryLimit,
                x.MonthlyReplyLimit
            })
            .FirstOrDefaultAsync(cancellationToken);

        int postLimit;
        int senderLimit;
        int receivedLimit;
        int replyLimit;

        if (paidEntitlement != null)
        {
            response.PackageCode = paidEntitlement.PackageCode;
            response.PackageName = paidEntitlement.Name;
            response.EntitlementStatus = paidEntitlement.Status;
            response.StartDate = paidEntitlement.StartDate;
            response.EndDate = paidEntitlement.EndDate;
            response.IsPaidPackage = true;

            postLimit = paidEntitlement.MonthlyAdvertisementPostLimit;
            senderLimit = paidEntitlement.MonthlySenderEnquiryLimit;
            receivedLimit = paidEntitlement.MonthlyAdvertisementEnquiryLimit;
            replyLimit = paidEntitlement.MonthlyReplyLimit;
        }
        else
        {
            var freePackage = await _context.EnquiryPackages
                .AsNoTracking()
                .Where(x => x.PackageCode == "FREE" && x.IsActive)
                .Select(x => new
                {
                    x.PackageCode,
                    x.Name,
                    x.MonthlyAdvertisementPostLimit,
                    x.MonthlySenderEnquiryLimit,
                    x.MonthlyAdvertisementEnquiryLimit,
                    x.MonthlyReplyLimit
                })
                .FirstOrDefaultAsync(cancellationToken);

            if (freePackage == null)
            {
                throw new InvalidOperationException(
                    "The FREE package is unavailable. Please contact support.");
            }

            response.PackageCode = freePackage.PackageCode;
            response.PackageName = freePackage.Name;
            response.EntitlementStatus = "FREE";
            response.IsPaidPackage = false;

            postLimit = freePackage.MonthlyAdvertisementPostLimit;
            senderLimit = freePackage.MonthlySenderEnquiryLimit;
            receivedLimit = freePackage.MonthlyAdvertisementEnquiryLimit;
            replyLimit = freePackage.MonthlyReplyLimit;
        }

        var postsUsed = await _context.AuditLogs
            .AsNoTracking()
            .Where(x =>
                x.UserID == userId &&
                x.Action == "ADVERTISEMENT_SUBMITTED" &&
                x.EntityName == "Advertisement" &&
                x.EntityID != null &&
                x.CreatedDate >= periodStart &&
                x.CreatedDate < periodEnd)
            .Select(x => x.EntityID)
            .Distinct()
            .CountAsync(cancellationToken);

        var enquiriesSent = await _context.AdvertisementEnquiries
            .AsNoTracking()
            .CountAsync(x =>
                x.SenderUserID == userId &&
                x.CreatedDate >= periodStart &&
                x.CreatedDate < periodEnd,
                cancellationToken);

        var repliesSent = await _context.AdvertisementEnquiryMessages
            .AsNoTracking()
            .CountAsync(x =>
                x.SenderUserID == userId &&
                x.CreatedDate >= periodStart &&
                x.CreatedDate < periodEnd,
                cancellationToken);

        response.AdvertisementPosts = new PackageUsageBalance
        {
            Limit = postLimit,
            Used = postsUsed
        };

        response.EnquiriesSent = new PackageUsageBalance
        {
            Limit = senderLimit,
            Used = enquiriesSent
        };

        response.RepliesSent = new PackageUsageBalance
        {
            Limit = replyLimit,
            Used = repliesSent
        };

        var ownedAdvertisements = await _context.Advertisements
            .AsNoTracking()
            .Where(x => x.UserID == userId)
            .Select(x => new
            {
                x.AdvertisementID,
                x.Title
            })
            .ToListAsync(cancellationToken);

        var ownedAdvertisementIds = ownedAdvertisements
            .Select(x => x.AdvertisementID)
            .ToList();

        var receivedCounts = ownedAdvertisementIds.Count == 0
            ? new List<(long AdvertisementID, int Used)>()
            : (await _context.AdvertisementEnquiries
                .AsNoTracking()
                .Where(x =>
                    ownedAdvertisementIds.Contains(x.AdvertisementID) &&
                    x.CreatedDate >= periodStart &&
                    x.CreatedDate < periodEnd)
                .GroupBy(x => x.AdvertisementID)
                .Select(g => new
                {
                    AdvertisementID = g.Key,
                    Used = g.Count()
                })
                .ToListAsync(cancellationToken))
                .Select(x => (x.AdvertisementID, x.Used))
                .ToList();

        var receivedCountByAdvertisement = receivedCounts.ToDictionary(
            x => x.AdvertisementID,
            x => x.Used);

        response.AdvertisementEnquiriesReceived = ownedAdvertisements
            .Select(advertisement => new AdvertisementEnquiryUsageBalance
            {
                AdvertisementID = advertisement.AdvertisementID,
                AdvertisementTitle = advertisement.Title,
                Limit = receivedLimit,
                Used = receivedCountByAdvertisement.TryGetValue(
                    advertisement.AdvertisementID,
                    out var used)
                    ? used
                    : 0
            })
            .OrderBy(x => x.AdvertisementTitle)
            .ToList();

        response.QueuedPackages = await _context.UserPackageEntitlements
            .AsNoTracking()
            .Where(x =>
                x.UserID == userId &&
                x.Status == UserPackageEntitlementStatuses.Pending &&
                x.StartDate > now &&
                x.EnquiryPackage.PackageCode != "FREE" &&
                x.Payments.Any(p =>
                    p.Status == PaymentTransactionStatuses.Confirmed &&
                    p.ConfirmedDate != null &&
                    p.RefundedDate == null))
            .OrderBy(x => x.StartDate)
            .Select(x => new QueuedPackageBalance
            {
                PackageCode = x.EnquiryPackage.PackageCode,
                PackageName = x.EnquiryPackage.Name,
                EntitlementStatus = x.Status,
                StartDate = x.StartDate,
                EndDate = x.EndDate,
                MonthlyAdvertisementPostLimit = x.MonthlyAdvertisementPostLimit,
                MonthlySenderEnquiryLimit = x.MonthlySenderEnquiryLimit,
                MonthlyAdvertisementEnquiryLimit = x.MonthlyAdvertisementEnquiryLimit,
                MonthlyReplyLimit = x.MonthlyReplyLimit
            })
            .ToListAsync(cancellationToken);
        return response;
    }

public async Task<PackagePurchaseRequestResponse>
    CreatePurchaseRequestAsync(
        long userId,
        int enquiryPackageId,
        CancellationToken cancellationToken = default)
{
    if (userId <= 0)
    {
        throw new InvalidOperationException(
            "The authenticated user ID is invalid.");
    }

    var paidPackagesEnabled =
        await _settings.GetBoolAsync(
            "PAID_PACKAGES_ENABLED",
            cancellationToken) ?? false;

    if (!paidPackagesEnabled)
    {
        throw new InvalidOperationException(
            "Paid package purchases are currently disabled.");
    }

    var package = await _context.EnquiryPackages
        .AsNoTracking()
        .SingleOrDefaultAsync(
            x => x.EnquiryPackageID == enquiryPackageId &&
                 x.IsActive,
            cancellationToken);

    if (package == null)
    {
        throw new InvalidOperationException(
            "The requested package does not exist or is inactive.");
    }

    if (string.Equals(
        package.PackageCode,
        "FREE",
        StringComparison.OrdinalIgnoreCase) ||
    package.Price <= 0m ||
    (package.MonthlySenderEnquiryLimit <= 0 &&
     package.MonthlyAdvertisementEnquiryLimit <= 0 &&
     package.MonthlyReplyLimit <= 0 &&
     package.MonthlyAdvertisementPostLimit <= 0))
{
    throw new InvalidOperationException(
        "This package is not configured for paid purchase.");
}

    await using var transaction =
        await _context.Database.BeginTransactionAsync(
            System.Data.IsolationLevel.Serializable,
            cancellationToken);

    var hasOutstandingRequest =
        await _context.PackagePurchaseRequests
            .AnyAsync(
                x => x.UserID == userId &&
                     (x.Status == PackagePurchaseRequestStatuses.Pending ||
                      x.Status == PackagePurchaseRequestStatuses.Approved),
                cancellationToken);

    if (hasOutstandingRequest)
    {
        throw new InvalidOperationException(
            "You already have an outstanding package purchase request. " +
            "Complete or resolve it before requesting another package.");
    }

    var request = new PackagePurchaseRequest
    {
        UserID = userId,
        EnquiryPackageID = package.EnquiryPackageID,
        Amount = package.Price,
        CurrencyCode = package.CurrencyCode,
        Status = PackagePurchaseRequestStatuses.Pending,
        CreatedDate = DateTime.UtcNow
    };

    _context.PackagePurchaseRequests.Add(request);

    await _context.SaveChangesAsync(cancellationToken);
    await transaction.CommitAsync(cancellationToken);

    request.EnquiryPackage = package;

    return MapPurchaseRequest(request);
}

    public async Task<AdminPackagePurchaseRequestResponse?>
        CancelPurchaseRequestAsync(
            long adminUserId,
            long purchaseRequestId,
            CancellationToken cancellationToken = default)
    {
        if (adminUserId <= 0)
        {
            throw new InvalidOperationException(
                "The authenticated administrator ID is invalid.");
        }

        if (purchaseRequestId <= 0)
        {
            throw new InvalidOperationException(
                "The purchase request ID is invalid.");
        }

        await using var transaction =
            await _context.Database.BeginTransactionAsync(
                System.Data.IsolationLevel.Serializable,
                cancellationToken);

        var request = await _context.PackagePurchaseRequests
            .Include(x => x.EnquiryPackage)
            .SingleOrDefaultAsync(
                x => x.PackagePurchaseRequestID == purchaseRequestId,
                cancellationToken);

        if (request is null)
        {
            return null;
        }

        if (request.Status != PackagePurchaseRequestStatuses.Pending &&
            request.Status != PackagePurchaseRequestStatuses.Approved)
        {
            throw new InvalidOperationException(
                "Only pending or approved unpaid purchase requests can be cancelled.");
        }

        var now = DateTime.UtcNow;
        var oldStatus = request.Status;

        request.Status = PackagePurchaseRequestStatuses.Cancelled;
        request.ModifiedDate = now;

        _context.AuditLogs.Add(new AuditLog
        {
            UserID = adminUserId,
            Action = "PACKAGE_PURCHASE_CANCELLED",
            EntityName = "PackagePurchaseRequest",
            EntityID = request.PackagePurchaseRequestID.ToString(),
            OldValues = JsonSerializer.Serialize(new
            {
                Status = oldStatus
            }),
            NewValues = JsonSerializer.Serialize(new
            {
                Status = request.Status,
                ModifiedDate = request.ModifiedDate
            }),
            CreatedDate = now
        });

        await _context.SaveChangesAsync(cancellationToken);
        await transaction.CommitAsync(cancellationToken);

        return MapAdminPurchaseRequest(request);
    }
    public async Task<AdminPackagePurchaseRequestResponse?>
ReviewPurchaseRequestAsync(
long adminUserId,
long purchaseRequestId,
bool approve,
CancellationToken cancellationToken = default)
{
if (adminUserId <= 0)
{
throw new InvalidOperationException(
"The authenticated administrator ID is invalid.");
}

if (purchaseRequestId <= 0)
{
    throw new InvalidOperationException(
        "The purchase request ID is invalid.");
}

await using var transaction =
    await _context.Database.BeginTransactionAsync(
        System.Data.IsolationLevel.Serializable,
        cancellationToken);

var request = await _context.PackagePurchaseRequests
    .Include(x => x.EnquiryPackage)
    .SingleOrDefaultAsync(
        x => x.PackagePurchaseRequestID == purchaseRequestId,
        cancellationToken);

if (request == null)
{
    return null;
}

if (request.Status != PackagePurchaseRequestStatuses.Pending)
{
    throw new InvalidOperationException(
        "Only pending purchase requests can be reviewed.");
}

var now = DateTime.UtcNow;
var oldStatus = request.Status;

var newStatus = approve
    ? PackagePurchaseRequestStatuses.Approved
    : PackagePurchaseRequestStatuses.Rejected;

var oldValues = new
{
    Status = oldStatus,
    request.ModifiedDate
};

request.Status = newStatus;
request.ModifiedDate = now;

var newValues = new
{
    Status = request.Status,
    request.ModifiedDate
};

_context.AuditLogs.Add(new AuditLog
{
    UserID = adminUserId,
    Action = approve
        ? "PACKAGE_PURCHASE_APPROVED"
        : "PACKAGE_PURCHASE_REJECTED",
    EntityName = "PackagePurchaseRequest",
    EntityID = request.PackagePurchaseRequestID.ToString(),
    OldValues = JsonSerializer.Serialize(oldValues),
    NewValues = JsonSerializer.Serialize(newValues),
    CreatedDate = now
});

await _context.SaveChangesAsync(cancellationToken);
await transaction.CommitAsync(cancellationToken);

// Approval permits the request to proceed to payment.
// It does not confirm payment or activate an entitlement.
return MapAdminPurchaseRequest(request);

}

    public async Task<AdminPackagePurchaseRequestResponse?>
ConfirmPackagePaymentAsync(
long adminUserId,
long purchaseRequestId,
ConfirmPackagePaymentRequest request,
CancellationToken cancellationToken = default)
{
if (adminUserId <= 0)
{
throw new InvalidOperationException(
"The authenticated administrator ID is invalid.");
}

if (purchaseRequestId <= 0)
{
    throw new InvalidOperationException(
        "The purchase request ID is invalid.");
}

ArgumentNullException.ThrowIfNull(request);

if (string.IsNullOrWhiteSpace(request.PaymentMethod))
{
    throw new InvalidOperationException(
        "Payment method is required.");
}

var paymentMethod =
    request.PaymentMethod.Trim().ToUpperInvariant();

if (paymentMethod != "BANK_TRANSFER" &&
    paymentMethod != "BENEFITPAY")
{
    throw new InvalidOperationException(
        "Payment method must be BANK_TRANSFER or BENEFITPAY.");
}

var settingKey = paymentMethod == "BANK_TRANSFER"
    ? "BANK_TRANSFER_ENABLED"
    : "BENEFITPAY_ENABLED";

var paymentMethodEnabled =
    await _settings.GetBoolAsync(
        settingKey,
        cancellationToken) ?? false;

if (!paymentMethodEnabled)
{
    throw new InvalidOperationException(
        "The selected payment method is currently disabled.");
}

var reference =
    request.ProviderTransactionReference?.Trim();

if (string.IsNullOrWhiteSpace(reference) ||
    reference.Length > 200)
{
    throw new InvalidOperationException(
        "A payment reference is required and must not exceed 200 characters.");
}

await using var transaction =
    await _context.Database.BeginTransactionAsync(
        System.Data.IsolationLevel.Serializable,
        cancellationToken);

var purchaseRequest = await _context.PackagePurchaseRequests
    .Include(x => x.EnquiryPackage)
    .SingleOrDefaultAsync(
        x => x.PackagePurchaseRequestID == purchaseRequestId,
        cancellationToken);

if (purchaseRequest == null)
{
    return null;
}

if (purchaseRequest.Status !=
    PackagePurchaseRequestStatuses.Approved)
{
    throw new InvalidOperationException(
        "Only approved purchase requests can have payment confirmed.");
}

var package = purchaseRequest.EnquiryPackage;

if (!package.IsActive ||
    string.Equals(
        package.PackageCode,
        "FREE",
        StringComparison.OrdinalIgnoreCase) ||
    purchaseRequest.Amount <= 0m)
{
    throw new InvalidOperationException(
        "The purchase request is not eligible for paid activation.");
}

    package.ValidityDays.Value <= 0 ||
    package.ValidityDays.Value > 3650)

        "The paid package must have a validity period between 1 and 3650 days before payment can be confirmed.");

    _context, purchaseRequest.UserID, now, cancellationToken);

{
    if (!entitlement.EndDate.HasValue)

        throw new InvalidOperationException(
            "A package without an expiry date cannot be queued behind.");
    }

    if (entitlement.EndDate.Value <= now)
    {
        entitlement.Status = UserPackageEntitlementStatuses.Expired;
        entitlement.ModifiedDate = now;
    }
}

await _context.SaveChangesAsync(cancellationToken);

var latestScheduledEnd = await _context.UserPackageEntitlements
    .Where(x =>
        x.UserID == purchaseRequest.UserID &&
        x.EndDate.HasValue &&
        (x.Status == UserPackageEntitlementStatuses.Active ||
         (x.Status == UserPackageEntitlementStatuses.Pending &&
          x.Payments.Any(p =>
              p.Status == PaymentTransactionStatuses.Confirmed &&
              p.ConfirmedDate.HasValue &&
              !p.RefundedDate.HasValue))))
    .MaxAsync(x => (DateTime?)x.EndDate, cancellationToken);

    latestScheduledEnd.HasValue && latestScheduledEnd.Value > now
        ? latestScheduledEnd.Value
        : now;
var newEntitlement = new UserPackageEntitlement
{

    EnquiryPackageID = package.EnquiryPackageID,

    // Copy the package configuration at activation time.

    Amount = purchaseRequest.Amount,
    CurrencyCode = purchaseRequest.CurrencyCode,
    PaymentMethod = paymentMethod,

        ? "MANUAL"
        : "BENEFITPAY",
    ProviderTransactionReference = reference,
    Status = PaymentTransactionStatuses.Confirmed,
    CreatedDate = now,
    ConfirmedDate = now
};

newEntitlement.Payments.Add(payment);
_context.UserPackageEntitlements.Add(newEntitlement);

var oldStatus = purchaseRequest.Status;

purchaseRequest.Status =
    PackagePurchaseRequestStatuses.Completed;

purchaseRequest.ModifiedDate = now;

// Save the entitlement and payment first so generated IDs are available.
// This save remains inside the transaction.
await _context.SaveChangesAsync(cancellationToken);

_context.AuditLogs.Add(new AuditLog
{
    UserID = adminUserId,
    Action = "PACKAGE_PAYMENT_CONFIRMED",
    EntityName = "PackagePurchaseRequest",
    EntityID = purchaseRequest.PackagePurchaseRequestID.ToString(),
    OldValues = JsonSerializer.Serialize(new
    {

    }),
    NewValues = JsonSerializer.Serialize(new
    {
        Status = purchaseRequest.Status,

        CurrencyCode = payment.CurrencyCode,
        UserPackageEntitlementID =
            newEntitlement.UserPackageEntitlementID,
        ConfirmedDate = payment.ConfirmedDate,
        ModifiedDate = purchaseRequest.ModifiedDate
    }),

});

// Persist the audit record before committing the same transaction.
await _context.SaveChangesAsync(cancellationToken);
await transaction.CommitAsync(cancellationToken);

return MapAdminPurchaseRequest(purchaseRequest);

}

    public async Task<bool?> RefundPackagePaymentAsync(
        long adminUserId,
        long paymentTransactionId,
        RefundPackagePaymentRequest request,
        CancellationToken cancellationToken = default)
    {
        if (adminUserId <= 0)
        {
            throw new InvalidOperationException(
                "A valid administrator identity is required.");
        }

        if (paymentTransactionId <= 0)
        {
            throw new InvalidOperationException(
                "A valid payment transaction ID is required.");
        }

        if (request is null || !request.RefundProcessedExternally)
        {
            throw new InvalidOperationException(
                "Confirm that the external refund has already been processed.");
        }

        var refundReference = request.ExternalRefundReference?.Trim();
        var reason = request.Reason?.Trim();

        if (string.IsNullOrWhiteSpace(refundReference) ||
            refundReference.Length > 200)
        {
            throw new InvalidOperationException(
                "A valid external refund reference is required (maximum 200 characters).");
        }

        if (string.IsNullOrWhiteSpace(reason) ||
            reason.Length < 5 ||
            reason.Length > 1000)
        {
            throw new InvalidOperationException(
                "The refund reason must be between 5 and 1000 characters.");
        }

        await using var transaction =
            await _context.Database.BeginTransactionAsync(
                System.Data.IsolationLevel.Serializable,
                cancellationToken);

        var payment = await _context.PaymentTransactions
            .Include(x => x.UserPackageEntitlement)
                .ThenInclude(x => x.Payments)
            .SingleOrDefaultAsync(
                x => x.PaymentTransactionID == paymentTransactionId,
                cancellationToken);

        if (payment is null)
        {
            return null;
        }

        var refundReferenceInUse = await _context.PaymentTransactions
            .AnyAsync(x =>
                x.PaymentTransactionID != paymentTransactionId &&
                x.ExternalRefundReference == refundReference,
                cancellationToken);

        if (refundReferenceInUse)
        {
            throw new InvalidOperationException(
                "This external refund reference has already been recorded.");
        }

        if (payment.Status != PaymentTransactionStatuses.Confirmed ||
            !payment.ConfirmedDate.HasValue ||
            payment.RefundedDate.HasValue)
        {
            throw new InvalidOperationException(
                "Only a confirmed payment that has not already been refunded can be refunded.");
        }

        var entitlement = payment.UserPackageEntitlement;
        var now = DateTime.UtcNow;

        var oldPaymentStatus = payment.Status;
        var oldRefundedDate = payment.RefundedDate;
        var oldEntitlementStatus = entitlement.Status;

        payment.Status = PaymentTransactionStatuses.Refunded;
        payment.RefundedDate = now;
        payment.ExternalRefundReference = refundReference;

        var hasOtherValidPayment = entitlement.Payments.Any(other =>
            other.PaymentTransactionID != payment.PaymentTransactionID &&
            other.Status == PaymentTransactionStatuses.Confirmed &&
            other.ConfirmedDate.HasValue &&
            !other.RefundedDate.HasValue);

        if (!hasOtherValidPayment &&
            entitlement.Status != UserPackageEntitlementStatuses.Expired)
        {
            entitlement.Status = UserPackageEntitlementStatuses.Cancelled;
        }

        entitlement.ModifiedDate = now;

        _context.AuditLogs.Add(new AuditLog
        {
            UserID = adminUserId,
            Action = "PACKAGE_PAYMENT_REFUNDED",
            EntityName = "PaymentTransaction",
            EntityID = payment.PaymentTransactionID.ToString(),
            OldValues = JsonSerializer.Serialize(new
            {
                Status = oldPaymentStatus,
                RefundedDate = oldRefundedDate,
                EntitlementStatus = oldEntitlementStatus
            }),
            NewValues = JsonSerializer.Serialize(new
            {
                Status = payment.Status,
                RefundedDate = payment.RefundedDate,
                ExternalRefundReference = refundReference,
                Reason = reason,
                Amount = payment.Amount,
                CurrencyCode = payment.CurrencyCode,
                UserPackageEntitlementID =
                    entitlement.UserPackageEntitlementID,
                EntitlementStatus = entitlement.Status,
                ModifiedDate = entitlement.ModifiedDate
            }),
            CreatedDate = now
        });

        await _context.SaveChangesAsync(cancellationToken);
        await transaction.CommitAsync(cancellationToken);

        return true;
    }

using HRInfoAdvertisements.Domain.Billing;
using HRInfoAdvertisements.Infrastructure.Data;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Storage;

namespace HRInfoAdvertisements.Infrastructure.Billing;

internal static class PackageEntitlementActivation
{
    public static async Task ActivateDueAsync(
        ApplicationDbContext context,
        long userId,
        DateTime now,
        CancellationToken cancellationToken = default)
    {
        var transaction = context.Database.CurrentTransaction
            ?? throw new InvalidOperationException(
                "Package activation requires an active database transaction.");

        await using var command =
            context.Database.GetDbConnection().CreateCommand();

        command.Transaction = transaction.GetDbTransaction();
        command.CommandText = """
            DECLARE @result int;
            EXEC @result = sp_getapplock
                @Resource = @resource,
                @LockMode = 'Exclusive',
                @LockOwner = 'Transaction',
                @LockTimeout = 10000;
            SELECT @result;
            """;

        var parameter = command.CreateParameter();
        parameter.ParameterName = "@resource";
        parameter.Value =
            $"HRInfoAdvertisements:PackageActivation:User:{userId}";
        command.Parameters.Add(parameter);

        var lockResult = Convert.ToInt32(
            await command.ExecuteScalarAsync(cancellationToken));

        if (lockResult < 0)
            throw new InvalidOperationException(
                "Could not acquire the package activation lock.");

        while (true)
        {
            var expired = await context.UserPackageEntitlements
                .Where(x =>
                    x.UserID == userId &&
                    x.Status == UserPackageEntitlementStatuses.Active &&
                    x.EndDate.HasValue &&
                    x.EndDate.Value <= now)
                .ToListAsync(cancellationToken);

            foreach (var item in expired)
            {
                item.Status = UserPackageEntitlementStatuses.Expired;
                item.ModifiedDate = now;
            }

            if (expired.Count > 0)
                await context.SaveChangesAsync(cancellationToken);

            var hasActive = await context.UserPackageEntitlements
                .AnyAsync(x =>
                    x.UserID == userId &&
                    x.Status == UserPackageEntitlementStatuses.Active &&
                    x.StartDate <= now &&
                    (!x.EndDate.HasValue || x.EndDate.Value > now),
                    cancellationToken);

            if (hasActive)
                return;

            var next = await context.UserPackageEntitlements
                .Where(x =>
                    x.UserID == userId &&
                    x.Status == UserPackageEntitlementStatuses.Pending &&
                    x.StartDate <= now &&
                    x.EnquiryPackage.PackageCode != "FREE" &&
                    x.Payments.Any(p =>
                        p.Status == PaymentTransactionStatuses.Confirmed &&
                        p.ConfirmedDate.HasValue &&
                        !p.RefundedDate.HasValue))
                .OrderBy(x => x.StartDate)
                .ThenBy(x => x.UserPackageEntitlementID)
                .FirstOrDefaultAsync(cancellationToken);

            if (next == null)
                return;

            next.Status = UserPackageEntitlementStatuses.Active;
            next.ModifiedDate = now;
            await context.SaveChangesAsync(cancellationToken);

            // If an old queued package has already expired, advance again.
            if (next.EndDate.HasValue && next.EndDate.Value <= now)
            {
                next.Status = UserPackageEntitlementStatuses.Expired;
                next.ModifiedDate = now;
                await context.SaveChangesAsync(cancellationToken);
                continue;
            }

            return;
        }
    }
}

        if (maxPrice.HasValue)
        {
            query = query.Where(x =>
                x.Price.HasValue &&
                x.Price.Value <= maxPrice.Value);
        }

        // --------------------------------------------------------
        // Count
        // --------------------------------------------------------

        var totalRecords =
            await query.CountAsync();

        // --------------------------------------------------------
        // Execute Query
        // --------------------------------------------------------

        var items =
            await query
                .OrderByDescending(x => x.IsFeatured)
                .ThenByDescending(x => x.PublishedDate)
                .ThenByDescending(x => x.CreatedDate)
                .Skip((pageNumber - 1) * pageSize)
                .Take(pageSize)
                .ToListAsync();

        return new AdvertisementListResponse
        {
            Items =
                items
                    .Select(x =>
                        MapToResponse(
                            x,
                            includePrivateContact: false))
                    .ToList(),

            PageNumber =
                pageNumber,

            PageSize =
                pageSize,

            TotalRecords =
                totalRecords,

            TotalPages =
                (int)Math.Ceiling(
                    totalRecords /
                    (double)pageSize)
        };
    }

    public async Task<bool> SubmitAsync(
    long userId,
    long advertisementId)
{
    await using var transaction =
        await _context.Database.BeginTransactionAsync(
            System.Data.IsolationLevel.Serializable);

    var advertisement =
        await _context.Advertisements
            .Include(x => x.Status)
            .FirstOrDefaultAsync(x =>
                x.AdvertisementID == advertisementId &&
                x.UserID == userId);

    if (advertisement == null)
        return false;

    if (advertisement.Status == null ||
        advertisement.Status.StatusCode != "DRAFT")
        return false;

    if (string.IsNullOrWhiteSpace(advertisement.Title) ||
        string.IsNullOrWhiteSpace(advertisement.Description))
        return false;

    var hasWhatsApp =
        !string.IsNullOrWhiteSpace(advertisement.WhatsAppNumber);

    var hasEmail =
        !string.IsNullOrWhiteSpace(advertisement.ContactEmail);

    var hasPublicWhatsApp =
        hasWhatsApp && advertisement.ShowWhatsAppToPublic;

    var hasPublicEmail =
        hasEmail && advertisement.ShowEmailToPublic;

    if (!hasWhatsApp && !hasEmail)
        throw new InvalidOperationException(
            "At least one contact method is required before submitting the advertisement.");

    if (!hasPublicWhatsApp && !hasPublicEmail)
        throw new InvalidOperationException(
            "At least one contact method must be enabled for public visibility before submitting the advertisement.");

    var now = DateTime.UtcNow;
    var monthStart = new DateTime(now.Year, now.Month, 1);
    var nextMonthStart = monthStart.AddMonths(1);
    var monthKey = monthStart.ToString("yyyyMM");

await AcquireAdvertisementSubmissionLockAsync(
    $"HRInfoAdvertisements:AdvertisementSubmission:User:{userId}:{monthKey}");

    await PackageEntitlementActivation.ActivateDueAsync(_context, userId, now);

    var entityId = advertisementId.ToString();

    var alreadyCounted =
        await _context.AuditLogs.AnyAsync(x =>
            x.UserID == userId &&
            x.Action == "ADVERTISEMENT_SUBMITTED" &&
            x.EntityName == "Advertisement" &&
            x.EntityID == entityId &&
            x.CreatedDate >= monthStart &&
            x.CreatedDate < nextMonthStart);

    if (!alreadyCounted)
{
    // Prefer the user's currently active paid package entitlement.
    
var paidPackageLimit =
    await _context.UserPackageEntitlements
        .Where(x =>
            x.UserID == userId &&
            x.Status == UserPackageEntitlementStatuses.Active &&
            x.StartDate <= now &&
            (!x.EndDate.HasValue || x.EndDate.Value > now) &&
            x.EnquiryPackage.PackageCode != "FREE" &&
            x.Payments.Any(payment =>
                payment.Status == PaymentTransactionStatuses.Confirmed &&
                payment.ConfirmedDate != null &&
                payment.RefundedDate == null))
        .OrderByDescending(x => x.StartDate)
        .Select(x => (int?)x.MonthlyAdvertisementPostLimit)
        .FirstOrDefaultAsync();

    int monthlyPostLimit;

    if (paidPackageLimit.HasValue)
    {
        monthlyPostLimit = paidPackageLimit.Value;
    }
    else
    {
        // No active paid entitlement: use the configured FREE allowance.
        var freePackageLimit =
            await _context.EnquiryPackages
                .Where(x =>
                    x.PackageCode == "FREE" &&
                    x.IsActive)
                .Select(x => (int?)x.MonthlyAdvertisementPostLimit)
                .FirstOrDefaultAsync();

        if (!freePackageLimit.HasValue)
        {
            throw new InvalidOperationException(
                "The active FREE package is not configured. Please contact the administrator.");
        }

        monthlyPostLimit = freePackageLimit.Value;
    }

    var monthlyCount =
        await _context.AuditLogs
            .Where(x =>
                x.UserID == userId &&
                x.Action == "ADVERTISEMENT_SUBMITTED" &&
                x.EntityName == "Advertisement" &&
                x.CreatedDate >= monthStart &&
                x.CreatedDate < nextMonthStart)
            .Select(x => x.EntityID)
            .Distinct()
            .CountAsync();

    if (monthlyCount >= monthlyPostLimit)
    {
        throw new InvalidOperationException(
            $"You have reached your monthly advertisement submission limit of {monthlyPostLimit}. Please upgrade your package or contact the administrator.");
    }
}

    var pendingReviewStatus =
        await _context.AdvertisementStatuses
            .FirstOrDefaultAsync(x =>
                x.StatusCode == "PENDING_REVIEW" &&
                x.IsActive);

    if (pendingReviewStatus == null)
        throw new InvalidOperationException(
            "Advertisement PENDING_REVIEW status was not found or is inactive.");

    advertisement.StatusID = pendingReviewStatus.StatusID;
    advertisement.ModifiedDate = now;
    advertisement.ModifiedBy = userId;

    if (!alreadyCounted)
    {
        _context.AuditLogs.Add(new AuditLog
        {
            UserID = userId,
            Action = "ADVERTISEMENT_SUBMITTED",
            EntityName = "Advertisement",
            EntityID = entityId,
            CreatedDate = now
        });
    }

    await _context.SaveChangesAsync();
    await transaction.CommitAsync();

    return true;
}

    private async Task<string>
        GenerateAdvertisementNumberAsync()
    {
        var year =
            DateTime.UtcNow.Year;

        var prefix =
            $"ADV-{year}-";

        var lastNumber =
            await _context.Advertisements
                .Where(x =>
                    x.AdvertisementNumber.StartsWith(
                        prefix))
                .OrderByDescending(x =>
                    x.AdvertisementID)
                .Select(x =>
                    x.AdvertisementNumber)
                .FirstOrDefaultAsync();

        var nextNumber = 1;

        if (!string.IsNullOrWhiteSpace(lastNumber))
        {
            var numberPart =
                lastNumber.Replace(
                    prefix,
                    "");

            if (int.TryParse(
                    numberPart,
                    out var parsedNumber))
            {
                nextNumber =
                    parsedNumber + 1;
            }
        }

        return
            $"{prefix}{nextNumber:D6}";
    }

    private static void ValidatePublicContactFlags(
        string? whatsappNumber,
        string? contactEmail,
        bool showWhatsAppToPublic,
        bool showEmailToPublic)
    {
        if (showWhatsAppToPublic &&
            string.IsNullOrWhiteSpace(whatsappNumber))
        {
            throw new InvalidOperationException(
                "A WhatsApp number is required when WhatsApp visibility is enabled.");
        }

        if (showEmailToPublic &&
            string.IsNullOrWhiteSpace(contactEmail))
        {
            throw new InvalidOperationException(
                "An email address is required when email visibility is enabled.");
        }
    }

    private async Task AcquireAdvertisementSubmissionLockAsync(
    string resource)
{
    var transaction = _context.Database.CurrentTransaction
        ?? throw new InvalidOperationException(
            "An active database transaction is required to check the advertisement submission limit.");

    await using var command =
        _context.Database.GetDbConnection().CreateCommand();

    command.Transaction = transaction.GetDbTransaction();
    command.CommandType = System.Data.CommandType.Text;
    command.CommandText = """
        DECLARE @result int;

        EXEC @result = sys.sp_getapplock
            @Resource = @resource,
            @LockMode = 'Exclusive',
            @LockOwner = 'Transaction',
            @LockTimeout = 10000,
            @DbPrincipal = 'public';

        SELECT @result;
        """;

    var parameter = command.CreateParameter();
    parameter.ParameterName = "@resource";
    parameter.DbType = System.Data.DbType.String;
    parameter.Size = 255;
    parameter.Value = resource;
    command.Parameters.Add(parameter);

    var result = await command.ExecuteScalarAsync();
    var lockResult = Convert.ToInt32(result);

    if (lockResult < 0)
    {
        throw new InvalidOperationException(
            "The advertisement submission limit could not be checked because of concurrent activity. Please try again.");
    }
}

public async Task<AdvertisementEnquiryResponse> CreateAsync(
    long senderUserId,
    CreateAdvertisementEnquiryRequest request)
{
    if (request.AdvertisementID <= 0)
        throw new ArgumentException("AdvertisementID is required.");

    if (string.IsNullOrWhiteSpace(request.Subject))
        throw new ArgumentException("Subject is required.");

    if (string.IsNullOrWhiteSpace(request.Message))
        throw new ArgumentException("Message is required.");

    var advertisement = await _context.Advertisements
        .AsNoTracking()
        .FirstOrDefaultAsync(x =>
            x.AdvertisementID == request.AdvertisementID);

    if (advertisement == null)
        throw new KeyNotFoundException("Advertisement not found.");

    var status = await _context.AdvertisementStatuses
        .AsNoTracking()
        .FirstOrDefaultAsync(x =>
            x.StatusID == advertisement.StatusID);

    if (status == null ||
        !string.Equals(
            status.StatusCode,
            "PUBLISHED",
            StringComparison.OrdinalIgnoreCase))
    {
        throw new InvalidOperationException(
            "Enquiries can only be sent for published advertisements.");
    }

    if (advertisement.UserID == senderUserId)
    {
        throw new InvalidOperationException(
            "You cannot send an enquiry for your own advertisement.");
    }

    var sender = await _context.Users
        .AsNoTracking()
        .FirstOrDefaultAsync(x => x.UserID == senderUserId);

    if (sender == null)
        throw new KeyNotFoundException("Sender user not found.");

    var now = DateTime.UtcNow;
    var monthStart = new DateTime(
        now.Year, now.Month, 1, 0, 0, 0, DateTimeKind.Utc);
    var nextMonthStart = monthStart.AddMonths(1);
    var monthKey = monthStart.ToString("yyyyMM");

    var senderLock =
        $"HRInfoAdvertisements:EnquiryLimit:Sender:{senderUserId}:{monthKey}";

    var advertisementLock =
        $"HRInfoAdvertisements:EnquiryLimit:Advertisement:{advertisement.AdvertisementID}:{monthKey}";

    string? rejectionType = null;
    int rejectionCount = 0;
    int rejectionLimit = 0;
    string? rejectionPackageCode = null;
    long? createdEnquiryId = null;

    await using (var transaction =
        await _context.Database.BeginTransactionAsync())
    {
        bool transactionCompleted = false;
        try
        {
            // Acquire locks in a consistent order for every request.
            foreach (var resource in new[] { senderLock, advertisementLock }
                .OrderBy(x => x, StringComparer.Ordinal))
            {
                await AcquireEnquiryLimitLockAsync(resource);
            }

            // Activate any paid packages whose scheduled start has arrived.
            foreach (var packageUserId in new[] { senderUserId, advertisement.UserID }
                .Distinct()
                .OrderBy(x => x))
            {
                await PackageEntitlementActivation.ActivateDueAsync(
                    _context, packageUserId, now);
            }
            // Resolve package entitlements while the limit locks are held.
            var senderPackage =
                await ResolveEnquiryPackageAsync(senderUserId, now);

            var ownerPackage =
                await ResolveEnquiryPackageAsync(advertisement.UserID, now);

            var senderCount =
                await _context.AdvertisementEnquiries.CountAsync(x =>
                    x.SenderUserID == senderUserId &&
                    x.CreatedDate >= monthStart &&
                    x.CreatedDate < nextMonthStart);

            if (senderCount >= senderPackage.MonthlySenderEnquiryLimit)
            {
                rejectionType = "SENDER_MONTHLY_LIMIT";
                rejectionCount = senderCount;
                rejectionLimit = senderPackage.MonthlySenderEnquiryLimit;
                rejectionPackageCode = senderPackage.PackageCode;
            }
            else
            {
                var advertisementCount =
                    await _context.AdvertisementEnquiries.CountAsync(x =>
                        x.AdvertisementID == advertisement.AdvertisementID &&
                        x.CreatedDate >= monthStart &&
                        x.CreatedDate < nextMonthStart);

                if (advertisementCount >=
                    ownerPackage.MonthlyAdvertisementEnquiryLimit)

                    rejectionType = "ADVERTISEMENT_MONTHLY_LIMIT";
                    rejectionCount = advertisementCount;
                    rejectionLimit =
                        ownerPackage.MonthlyAdvertisementEnquiryLimit;
                    rejectionPackageCode = ownerPackage.PackageCode;
                }
            }

            if (rejectionType == null)
            {
                var enquiry = new AdvertisementEnquiry
                {
                    AdvertisementID = advertisement.AdvertisementID,
                    RecipientUserID = advertisement.UserID,
                    SenderUserID = senderUserId,
                    Subject = request.Subject.Trim(),
                    Message = request.Message.Trim(),
                    ContactMobile =

                            ? sender.MobileNo
                            : request.ContactMobile.Trim(),
                    ContactEmail =
                        string.IsNullOrWhiteSpace(request.ContactEmail)
                            ? sender.Email
                            : request.ContactEmail.Trim(),
                    Status = "OPEN",
                    CreatedDate = DateTime.UtcNow
                };

                _context.AdvertisementEnquiries.Add(enquiry);
                await _context.SaveChangesAsync();

                _context.AuditLogs.Add(new AuditLog
                {
                    UserID = senderUserId,
                    Action = "ENQUIRY_CREATED",
                    EntityName = "AdvertisementEnquiry",
                    EntityID = enquiry.AdvertisementEnquiryID.ToString(),
                    NewValues = JsonSerializer.Serialize(new
                    {
                        AdvertisementEnquiryID =

                        AdvertisementID = advertisement.AdvertisementID,
                        SenderUserID = senderUserId,
                        RecipientUserID = advertisement.UserID,
                        Subject = enquiry.Subject,
                        CreatedDate = enquiry.CreatedDate
                    }),
                    CreatedDate = DateTime.UtcNow
                });

                await _context.SaveChangesAsync();
                await transaction.CommitAsync();

                transactionCompleted = true;
createdEnquiryId = enquiry.AdvertisementEnquiryID;

                    enquiry.AdvertisementEnquiryID)
                    ?? throw new InvalidOperationException(
                        "Unable to retrieve the created enquiry.");
            }

            // Release the limit locks before persisting the rejection audit.
            await transaction.RollbackAsync();
            transactionCompleted = true;
        }
        catch
        {
            if (!transactionCompleted)
    {
        try
        {
            await transaction.RollbackAsync();
        }
        catch
        {
            // Preserve the original exception.
        }
    }

            throw;

    public async Task<AdvertisementEnquiryMessageResponse?> ReplyAsync(
    long userId,
    long enquiryId,
    ReplyAdvertisementEnquiryRequest request)
{
    if (string.IsNullOrWhiteSpace(request.Message))
        throw new ArgumentException("Message is required.");

    var enquiry = await _context.AdvertisementEnquiries
        .FirstOrDefaultAsync(x =>
            x.AdvertisementEnquiryID == enquiryId &&
            (x.SenderUserID == userId ||
             x.RecipientUserID == userId));

    if (enquiry == null)
        return null;

    if (string.Equals(
        enquiry.Status,
        "CLOSED",
        StringComparison.OrdinalIgnoreCase))
    {
        throw new InvalidOperationException(
            "This enquiry is closed and cannot receive new replies.");
    }

    var now = DateTime.UtcNow;
    var monthStart = new DateTime(
        now.Year, now.Month, 1, 0, 0, 0, DateTimeKind.Utc);
    var nextMonthStart = monthStart.AddMonths(1);
    var monthKey = monthStart.ToString("yyyyMM");

    var replyLock =
        $"HRInfoAdvertisements:ReplyLimit:User:{userId}:{monthKey}";

    string? packageCode = null;
    var currentCount = 0;
    var configuredLimit = 0;
    var limitReached = false;
    AdvertisementEnquiryMessage? savedMessage = null;

    await using (var transaction =
        await _context.Database.BeginTransactionAsync())
    {
        var transactionCompleted = false;

        try
        {
            await AcquireEnquiryLimitLockAsync(replyLock);
            await PackageEntitlementActivation.ActivateDueAsync(_context, userId, now);

            var package = await ResolveEnquiryPackageAsync(userId, now);
            packageCode = package.PackageCode;
            configuredLimit = package.MonthlyReplyLimit;

            currentCount = await _context.AdvertisementEnquiryMessages
                .CountAsync(x =>
                    x.SenderUserID == userId &&
                    x.CreatedDate >= monthStart &&
                    x.CreatedDate < nextMonthStart);

            if (currentCount >= configuredLimit)
            {
                limitReached = true;

                await transaction.RollbackAsync();
                transactionCompleted = true;
            }
            else
            {
                var message = new AdvertisementEnquiryMessage
                {
                    AdvertisementEnquiryID =
                        enquiry.AdvertisementEnquiryID,
                    SenderUserID = userId,
                    Message = request.Message.Trim(),
                    CreatedDate = now,
                    IsRead = false
                };

                _context.AdvertisementEnquiryMessages.Add(message);

                enquiry.LastRepliedDate = now;
                enquiry.Status = "OPEN";

                await _context.SaveChangesAsync();
                await transaction.CommitAsync();

                transactionCompleted = true;
                savedMessage = message;
            }
        }
        catch
        {
            if (!transactionCompleted)
            {
                try
                {
                    await transaction.RollbackAsync();
                }
                catch
                {
                    // Preserve the original exception.
                }
            }

            throw;
        }
    }

    if (limitReached)
    {
        // Persist the audit record after releasing the limit lock.
        _context.AuditLogs.Add(new AuditLog
        {
            UserID = userId,
            Action = "REPLY_LIMIT_REJECTED",
            EntityName = "AdvertisementEnquiry",
            EntityID = enquiryId.ToString(),
            NewValues = JsonSerializer.Serialize(new
            {
                UserID = userId,
                AdvertisementEnquiryID = enquiryId,
                LimitType = "USER_MONTHLY_REPLY_LIMIT",
                CurrentCount = currentCount,
                ConfiguredLimit = configuredLimit,
                PackageCode = packageCode,
                MonthStartUtc = monthStart
            }),
            CreatedDate = DateTime.UtcNow
        });

        await _context.SaveChangesAsync();

        throw new InvalidOperationException(
            $"Your monthly reply limit of {configuredLimit} has been reached.");
    }

    var sender = await _context.Users
        .AsNoTracking()
        .FirstOrDefaultAsync(x => x.UserID == userId);

    return new AdvertisementEnquiryMessageResponse
    {
        AdvertisementEnquiryMessageID =
            savedMessage!.AdvertisementEnquiryMessageID,
        AdvertisementEnquiryID =
            savedMessage.AdvertisementEnquiryID,
        SenderUserID = savedMessage.SenderUserID,
        SenderName = GetUserDisplayName(sender),
        Message = savedMessage.Message,
        CreatedDate = savedMessage.CreatedDate,
        IsRead = savedMessage.IsRead,
        ReadDate = savedMessage.ReadDate
    };
}

        await _context.SaveChangesAsync();

        return true;
    }

    
private async Task<ResolvedEnquiryPackage> ResolveEnquiryPackageAsync(
    long userId,
    DateTime now)
{
    var paidPackage = await _context.UserPackageEntitlements
        .AsNoTracking()
        .Where(x =>
            x.UserID == userId &&
            x.Status == UserPackageEntitlementStatuses.Active &&
            x.StartDate <= now &&
            (x.EndDate == null || x.EndDate > now) &&
            x.EnquiryPackage.PackageCode != "FREE" &&
            x.Payments.Any(p =>
                p.Status == PaymentTransactionStatuses.Confirmed &&
                p.ConfirmedDate != null &&
                p.RefundedDate == null))
        .OrderByDescending(x => x.StartDate)
        .Select(x => new ResolvedEnquiryPackage(
            x.EnquiryPackage.PackageCode,
            x.MonthlySenderEnquiryLimit,
            x.MonthlyAdvertisementEnquiryLimit,
            x.MonthlyReplyLimit))
        .FirstOrDefaultAsync();

    if (paidPackage != null)
        return paidPackage;

    var freePackage = await _context.EnquiryPackages
        .AsNoTracking()
        .Where(x =>
            x.PackageCode == "FREE" &&
            x.IsActive)
        .Select(x => new ResolvedEnquiryPackage(
            x.PackageCode,
            x.MonthlySenderEnquiryLimit,
            x.MonthlyAdvertisementEnquiryLimit,
            x.MonthlyReplyLimit))
        .FirstOrDefaultAsync();

    return freePackage ?? throw new InvalidOperationException(
        "The FREE enquiry package is unavailable. Please contact support.");
}

private sealed record ResolvedEnquiryPackage(
    string PackageCode,
    int MonthlySenderEnquiryLimit,
    int MonthlyAdvertisementEnquiryLimit,
    int MonthlyReplyLimit);


private async Task AcquireEnquiryLimitLockAsync(
    string resource)
{
    var transaction = _context.Database.CurrentTransaction
        ?? throw new InvalidOperationException(
            "An active database transaction is required for enquiry limit checks.");

    await using var command =
        _context.Database.GetDbConnection().CreateCommand();

    command.Transaction = transaction.GetDbTransaction();
    command.CommandType = CommandType.Text;
    command.CommandText = """
        DECLARE @result int;

        EXEC @result = sys.sp_getapplock
            @Resource = @resource,
            @LockMode = 'Exclusive',
            @LockOwner = 'Transaction',
            @LockTimeout = 10000,
            @DbPrincipal = 'public';

        SELECT @result;
        """;

    var parameter = command.CreateParameter();
    parameter.ParameterName = "@resource";
    parameter.DbType = DbType.String;
    parameter.Size = 255;
    parameter.Value = resource;
    command.Parameters.Add(parameter);

    var result = await command.ExecuteScalarAsync();
    var lockResult = Convert.ToInt32(result);

    // SQL Server returns a negative value if the lock
    // cannot be acquired. Never continue without the lock.
    if (lockResult < 0)
    {
        throw new InvalidOperationException(
            "The enquiry limit could not be checked because of concurrent activity. Please try again.");
    }
}

@code {
    private List<PackagePurchaseRequestResponse> requests = new();
    private PublicSiteSettingsResponse paymentSettings = new();
    private MyPackageBalanceResponse? packageBalance;
    private bool isLoading = true;
    private string? errorMessage;
    private string? balanceErrorMessage;

    protected override async Task OnInitializedAsync()
    {
        if (AuthenticationStateProvider.IsAuthenticated)
        {
            await LoadRequestsAsync();
        }
        else
        {
            isLoading = false;
        }
    }

    private async Task LoadRequestsAsync()
    {
        if (!AuthenticationStateProvider.IsAuthenticated)
        {
            isLoading = false;
            Navigation.NavigateTo("/?auth=signin");
            return;
        }

        isLoading = true;
        errorMessage = null;

        try
        {
            paymentSettings = await SiteSettings.GetAsync();
            requests = await BillingService.GetMyPurchaseRequestsAsync();

            await LoadPackageBalanceAsync();
        }
        catch (Exception ex)
        {
            errorMessage = ex.Message;
        }
        finally
        {
            isLoading = false;
        }
    }

    private async Task LoadPackageBalanceAsync()
    {
        if (!AuthenticationStateProvider.IsAuthenticated)
        {
            packageBalance = null;
            balanceErrorMessage = "Please sign in to view your package balance.";
            return;
        }

        try
        {
            packageBalance = await BillingService.GetMyPackageBalanceAsync();
            balanceErrorMessage = null;
        }
        catch (Exception ex)
        {
            packageBalance = null;
            balanceErrorMessage = ex.Message;
        }
    }

    private static string FormatPackageDate(DateTime? value) =>
        value.HasValue
            ? value.Value.ToLocalTime().ToString("dd MMM yyyy", CultureInfo.CurrentCulture)
            : "Not scheduled";

    private static string FormatUsageLimit(int limit) =>
        limit < 0
            ? "Unlimited"
            : limit.ToString("N0", CultureInfo.CurrentCulture);

    private static string FormatUsageCount(PackageUsageBalance usage) =>
        $"{usage.Used.ToString("N0", CultureInfo.CurrentCulture)} / {FormatUsageLimit(usage.Limit)}";

    private static string GetUsageRemainingText(PackageUsageBalance usage) =>
        usage.Unlimited
            ? "Unlimited allowance"
            : $"{usage.Remaining.ToString("N0", CultureInfo.CurrentCulture)} remaining";

    private static string GetRemainingText(int limit, int remaining) =>
        limit < 0
            ? "Unlimited allowance"
            : $"{remaining.ToString("N0", CultureInfo.CurrentCulture)} remaining";
    private static string NormalizeStatus(string? status) =>
        (status ?? string.Empty).Trim().ToUpperInvariant();

    private static string GetStatusLabel(string status) =>
        status switch
        {
            "PENDING" => "Pending",
            "PENDING_REVIEW" => "Pending review",
            "SUBMITTED" => "Submitted",
            "APPROVED" => "Approved",
            "COMPLETED" => "Completed",
            "PAID" => "Paid",
            "REJECTED" => "Rejected",
            _ => string.IsNullOrWhiteSpace(status)
                ? "Status unavailable"
                : CultureInfo.CurrentCulture.TextInfo.ToTitleCase(
                    status.Replace('_', ' ').ToLowerInvariant())
        };

    private static string GetStatusClass(string status) =>
        status switch
        {
            "PENDING" or "PENDING_REVIEW" or "SUBMITTED" => "status-pending",
            "APPROVED" => "status-approved",
            "COMPLETED" or "PAID" => "status-completed",
            "REJECTED" => "status-rejected",
            _ => "status-default"
        };
